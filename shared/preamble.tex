}
\newcommand{\resumeSubheadingSep}{\vspace{0pt}}
\newcommand{\resumeItem}[1]{

}

%                compact itemize used throughout the resume.

% \Rule    horizontal divider between sections.
\newcommand{\Rule}{\noindent\rule{\linewidth}{0.4pt}}
% \SecGap  the small negative space placed before each \Rule.
\newcommand{\SecGap}{\vspace{-0.8em}}
